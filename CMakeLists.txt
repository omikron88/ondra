cmake_minimum_required(VERSION 3.16)
project(Ondra VERSION 0.4.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ONDRA_BUILD_GUI "Build the SDL2 desktop frontend" ON)
option(ONDRA_FETCH_SDL2 "Download and build SDL2 if it is not installed" OFF)
option(ONDRA_STATIC "Build standalone Windows executables with static SDL2 and MinGW runtimes" OFF)
if(ONDRA_STATIC AND NOT (MINGW AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU"))
    message(FATAL_ERROR "ONDRA_STATIC requires Windows with MinGW GCC (MSYS2 UCRT64/MINGW64 or a MinGW cross toolchain).")
endif()
include(CTest)

# Upstream sources are vendored, unmodified, at the revisions in THIRD_PARTY.md.
add_library(z80 STATIC third_party/z80/Z80.c)
target_include_directories(z80 PUBLIC third_party/z80 third_party/zeta/API)
target_compile_definitions(z80 PUBLIC Z80_STATIC PRIVATE
    Z80_WITH_Q Z80_WITH_ZILOG_NMOS_LD_A_IR_BUG)
set_target_properties(z80 PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)

# Embed firmware and the original speaker recordings; no working-directory or
# installed-data-path dependency, and no Python/code generator needed.
set(resource_source "#include <ondra/resources.hpp>\n#include <stdexcept>\nnamespace ondra {\n")
file(GLOB resource_files CONFIGURE_DEPENDS assets/roms/*.rom assets/sound/*.sample)
if(ONDRA_BUILD_GUI)
    file(GLOB ui_resources CONFIGURE_DEPENDS assets/ui/*.rgba assets/ui/*.ttf)
    list(APPEND resource_files ${ui_resources})
endif()
set(resource_index 0)
set(resource_lookup "")
foreach(resource IN LISTS resource_files)
    get_filename_component(resource_name "${resource}" NAME)
    file(READ "${resource}" resource_hex HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," resource_bytes "${resource_hex}")
    string(APPEND resource_source "static const unsigned char r${resource_index}[] = {${resource_bytes}};\n")
    string(APPEND resource_lookup "if (name == \"${resource_name}\") return std::vector<std::uint8_t>(r${resource_index}, r${resource_index} + sizeof(r${resource_index}));\n")
    math(EXPR resource_index "${resource_index} + 1")
endforeach()
string(APPEND resource_source "std::vector<std::uint8_t> resource(const std::string& name) {\n${resource_lookup}throw std::runtime_error(\"Unknown embedded resource: \" + name);\n}\n}\n")
set(resource_previous "")
if(EXISTS "${CMAKE_CURRENT_BINARY_DIR}/resources.cpp")
    file(READ "${CMAKE_CURRENT_BINARY_DIR}/resources.cpp" resource_previous)
endif()
if(NOT resource_previous STREQUAL resource_source)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/resources.cpp" "${resource_source}")
endif()

add_library(ondra_core STATIC src/machine.cpp src/tape.cpp src/sound.cpp src/disassembler.cpp src/printer.cpp
    src/snapshot.cpp src/sdcard.cpp src/rtc.cpp src/files.cpp "${CMAKE_CURRENT_BINARY_DIR}/resources.cpp")
target_include_directories(ondra_core PUBLIC include)
target_link_libraries(ondra_core PUBLIC z80)
if(ONDRA_STATIC)
    # Propagate to all executables, including the headless runner and tests.
    # -static also selects libwinpthread.a; static-libstdc++ alone leaves it
    # as a DLL dependency with the POSIX-threaded MinGW toolchain.
    target_link_options(ondra_core INTERFACE -static -static-libgcc -static-libstdc++)
endif()
if(WIN32)
    target_link_libraries(ondra_core PUBLIC shell32)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(ondra_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(ondra-headless src/headless.cpp src/options.cpp)
target_link_libraries(ondra-headless PRIVATE ondra_core)

if(ONDRA_BUILD_GUI)
    if(ONDRA_STATIC)
        # Always build the pinned SDL source for a standalone EXE. A system
        # SDL2::SDL2 target commonly selects a DLL, even if libSDL2.a exists.
        message(STATUS "Standalone Windows build: building pinned SDL2 statically")
        include(cmake/SDL2.cmake)
        set(ondra_sdl_target SDL2::SDL2-static)
    else()
        find_package(SDL2 2.0.18 CONFIG QUIET)
        if(NOT TARGET SDL2::SDL2)
            find_package(PkgConfig QUIET)
            if(PkgConfig_FOUND)
                pkg_check_modules(SDL2 QUIET IMPORTED_TARGET sdl2>=2.0.18)
                if(TARGET PkgConfig::SDL2)
                    add_library(SDL2::SDL2 ALIAS PkgConfig::SDL2)
                endif()
            endif()
        endif()
        if(NOT TARGET SDL2::SDL2 AND ONDRA_FETCH_SDL2)
            include(cmake/SDL2.cmake)
        endif()
        if(NOT TARGET SDL2::SDL2)
            message(FATAL_ERROR "Install SDL2 development files, or set -DONDRA_FETCH_SDL2=ON. Use -DONDRA_BUILD_GUI=OFF for headless only.")
        endif()
        set(ondra_sdl_target SDL2::SDL2)
    endif()
    add_library(imgui STATIC third_party/imgui/imgui.cpp third_party/imgui/imgui_draw.cpp
        third_party/imgui/imgui_tables.cpp third_party/imgui/imgui_widgets.cpp
        third_party/imgui/backends/imgui_impl_sdl2.cpp third_party/imgui/backends/imgui_impl_sdlrenderer2.cpp)
    target_include_directories(imgui PUBLIC third_party/imgui third_party/imgui/backends)
    target_link_libraries(imgui PUBLIC ${ondra_sdl_target})
    add_library(ondra_ui STATIC src/ui.cpp)
    target_include_directories(ondra_ui PUBLIC include)
    target_compile_definitions(ondra_ui PRIVATE ONDRA_VERSION="${PROJECT_VERSION}")
    target_link_libraries(ondra_ui PUBLIC ondra_core imgui)
    if(WIN32)
        target_link_libraries(ondra_ui PRIVATE user32)
    endif()
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(ondra_ui PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_executable(ondra src/main.cpp src/options.cpp)
    target_compile_definitions(ondra PRIVATE SDL_MAIN_HANDLED)
    target_link_libraries(ondra PRIVATE ondra_ui)
    if(WIN32 AND NOT ONDRA_STATIC AND NOT TARGET PkgConfig::SDL2)
        get_target_property(sdl_type ${ondra_sdl_target} TYPE)
        if(sdl_type STREQUAL "SHARED_LIBRARY")
            add_custom_command(TARGET ondra POST_BUILD COMMAND ${CMAKE_COMMAND} -E
                copy_if_different "$<TARGET_FILE:${ondra_sdl_target}>" "$<TARGET_FILE_DIR:ondra>")
        endif()
    endif()
    install(TARGETS ondra RUNTIME DESTINATION bin)
    install(FILES third_party/imgui/LICENSE.txt DESTINATION share/doc/ondra/licenses/imgui)
    install(FILES assets/ui/LICENSE-Roboto assets/ui/NOTICE-Roboto DESTINATION share/doc/ondra/licenses/roboto)
    if(ONDRA_STATIC OR ONDRA_FETCH_SDL2)
        if(DEFINED sdl2_SOURCE_DIR AND EXISTS "${sdl2_SOURCE_DIR}/LICENSE.txt")
            install(FILES "${sdl2_SOURCE_DIR}/LICENSE.txt" DESTINATION share/doc/ondra/licenses/sdl2)
        endif()
    endif()
endif()

install(TARGETS ondra-headless RUNTIME DESTINATION bin)
install(FILES LICENSE THIRD_PARTY.md assets/roms/LICENSE-PLUS DESTINATION share/doc/ondra)
foreach(dependency z80 zeta)
    install(FILES third_party/${dependency}/COPYING third_party/${dependency}/COPYING.LESSER
        DESTINATION share/doc/ondra/licenses/${dependency})
endforeach()

if(BUILD_TESTING)

    add_executable(ondra-printer-tests tests/printer_tests.cpp)
    target_link_libraries(ondra-printer-tests PRIVATE ondra_core)
    add_test(NAME printers COMMAND ondra-printer-tests)
    add_executable(ondra-rtc-tests tests/rtc_tests.cpp src/options.cpp)
    target_link_libraries(ondra-rtc-tests PRIVATE ondra_core)

    if(ONDRA_BUILD_GUI AND NOT CMAKE_CROSSCOMPILING)
        add_test(NAME sdl-smoke COMMAND ondra --no-config --frames 5)
        set_tests_properties(sdl-smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
    endif()
    if(ONDRA_BUILD_GUI)
        add_executable(ondra-ui-tests tests/gui_tests.cpp src/options.cpp)
        target_link_libraries(ondra-ui-tests PRIVATE ondra_ui)
        if(NOT CMAKE_CROSSCOMPILING)
            add_test(NAME gui-interactions COMMAND ondra-ui-tests)
            set_tests_properties(gui-interactions PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
        endif()
    endif()
endif()

if(ONDRA_STATIC)
    if(NOT CMAKE_OBJDUMP)
        message(FATAL_ERROR "objdump is required to check standalone Windows executables. Install MinGW binutils.")
    endif()
    foreach(executable ondra ondra-headless ondra-tests ondra-ui-tests ondra-printer-tests ondra-rtc-tests)
        if(TARGET ${executable})
            set_property(TARGET ${executable} APPEND PROPERTY LINK_DEPENDS
                "${CMAKE_CURRENT_SOURCE_DIR}/cmake/CheckWindowsImports.cmake")
            add_custom_command(TARGET ${executable} POST_BUILD
                COMMAND ${CMAKE_COMMAND}
                    "-DONDRA_OBJDUMP=${CMAKE_OBJDUMP}"
                    "-DONDRA_EXECUTABLE=$<TARGET_FILE:${executable}>"
                    -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/CheckWindowsImports.cmake"
                VERBATIM)
        endif()
    endforeach()
endif()
